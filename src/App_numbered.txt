   402	          <span className="text-sm font-bold">대시보드</span>
   403	        </button>
   404	        <button 
   405	          onClick={() => setActiveTab('history')}
   406	          className={cn("flex flex-col items-center justify-center py-2 px-6 rounded-[24px] transition-all", activeTab === 'history' ? "bg-[#FAF5E8] text-[#82C881]" : "text-[#A89E95] hover:bg-[#FAF5E8]/50")}
   407	        >
   408	          <span className="text-2xl mb-1">{activeTab === 'history' ? '📖' : '📘'}</span>
   409	          <span className="text-sm font-bold">히스토리</span>
   410	        </button>
   411	      </nav>
   412	
   413	      {/* Toast */}
   414	      {toast && (
   415	        <div className="fixed top-24 left-1/2 -translate-x-1/2 z-50 bg-[#FFB067] text-white font-bold px-6 py-3 rounded-[24px] border-4 border-[#FFD5A8] shadow-lg animate-in slide-in-from-top-4 fade-in">
   416	          {toast}
   417	        </div>
   418	      )}
   419	
   420	      {/* Modals */}
   421	      <AddFeedingModal 
   422	        isOpen={isFeedingModalOpen} 
   423	        onClose={() => setFeedingModalOpen(false)} 
   424	        geckoId={activeGeckoId!}
   425	        onSuccess={(item: HistoryItem) => {
   426	          setHistory(prev => [item, ...prev].sort((a,b) => new Date(b.type==='metric'?b.recorded_at:b.fed_at).getTime() - new Date(a.type==='metric'?a.recorded_at:a.fed_at).getTime()));
   427	          showToast('피딩이 기록되었습니다.');
   428	          setFeedingModalOpen(false);
   429	        }}
   430	      />
   431	
   432	      <AddMetricModal 
   433	        isOpen={isMetricModalOpen} 
   434	        onClose={() => setMetricModalOpen(false)} 
   435	        geckoId={activeGeckoId!}
   436	        onSuccess={(item: HistoryItem) => {
   437	          setHistory(prev => [item, ...prev].sort((a,b) => new Date(b.type==='metric'?b.recorded_at:b.fed_at).getTime() - new Date(a.type==='metric'?a.recorded_at:a.fed_at).getTime()));
   438	          showToast('계측이 기록되었습니다.');
   439	          setMetricModalOpen(false);
   440	        }}
   441	      />
   442	
   443	      <AddGeckoModal 
   444	        isOpen={isAddGeckoModalOpen} 
   445	        onClose={() => setAddGeckoModalOpen(false)} 
   446	        onSuccess={(g: Gecko) => {
   447	          setGeckos(prev => [...prev, g]);
   448	          setActiveGeckoId(g.id);
   449	          showToast('새 개체가 등록되었습니다.');
   450	          setAddGeckoModalOpen(false);
   451	        }}
   452	      />
   453	
   454	      <Modal isOpen={!!itemToDelete} onClose={() => setItemToDelete(null)} title="기록 삭제">
   455	        <div className="space-y-6">
   456	          <p className="text-[#5C4D43] text-lg text-center mt-4">정말로 이 기록을 삭제하시겠습니까?</p>
   457	          <div className="flex gap-4">
   458	            <Button variant="outline" className="flex-1" onClick={() => setItemToDelete(null)}>
   459	              취소
   460	            </Button>
   461	            <Button className="flex-1 bg-red-500 hover:bg-red-600 text-white border-red-600" onClick={confirmDelete}>
   462	              삭제하기
   463	            </Button>
   464	          </div>
   465	        </div>
   466	      </Modal>
   467	    </div>
   468	  );
   469	}
